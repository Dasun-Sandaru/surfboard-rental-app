/users_global (Collection)          // Global user directory
    /{userId}
        name
        email
        role                       // "admin" | "staff"
        shop_id                    // Assigned shop
        phone
        is_active                  // for disabling accounts
        created_at


/shops (Collection)
    /{shopId}
        name
        location
        contact_number
        created_date
        owner_admin_uid            // First admin of the shop

        --------------------------------------------------------------------

        /users (Subcollection)     // Admin creates staff here
            /{userId}
                name
                email
                role               // "admin" or "staff"
                phone
                is_active
                created_at

        --------------------------------------------------------------------

        /customers (Subcollection)   // Staff create customers here
            /{customerId}
                first_name
                last_name
                phone
                nic
                email
                notes
                created_at

        --------------------------------------------------------------------

        /inventory (Subcollection)
            /{itemId}
                name
                brand
                size
                color
                type
                status              // available | rented | maintenance
                damage_fee_rule     // rule or fixed amount
                rental_rate_hour
                rental_rate_day
                notes
                created_at

        --------------------------------------------------------------------

        /agreement_templates (Subcollection)
            /{templateId}
                template_name
                content             // text or HTML
                updated_at

        --------------------------------------------------------------------

        /rentals (Subcollection)
            /{rentalId}
                customer_id
                item_id
                staff_id            // who created rental
                start_time
                expected_return_time
                actual_return_time  // null until returned
                status              // active | completed | overdue
                amount_expected
                amount_paid
                payment_status      // unpaid | partial | paid
                agreement_id        // digital signature file ref
                created_at

                ------------------------------------------------------------

                /payments (Subcollection)
                    /{paymentId}
                        amount
                        type         // advance | full | late_fee | damage_fee
                        method       // cash | card | online
                        timestamp
                        handled_by   // staff

                ------------------------------------------------------------

                /returns (Subcollection)
                    /{returnId}
                        return_time
                        late_fee
                        damage_fee
                        total_due
                        handled_by
                        notes

                ------------------------------------------------------------

                /damage_reports (Subcollection)
                    /{damageId}
                        description
                        severity      // minor | major
                        estimated_fee
                        created_at
                        reported_by   // staff

                        /photos (Subcollection)
                            /{photoId}
                                url
                                uploaded_at

        --------------------------------------------------------------------

        /activity_logs (Subcollection)
            /{logId}
                user_id
                action_type          // rental_created | return_processed |
                                    // staff_added | customer_added |
                details
                timestamp

        --------------------------------------------------------------------

        /reports (Subcollection)   // Optional (for exporting PDF/CSV)
            /{reportId}
                type                // income_report | damage_report | etc.
                date_range
                generated_by
                created_at
                file_url            // stored in Firebase Storage
